특히 나는 C. 발음 오류 탐정과 E. 목소리로 움직이는 모험이 마음에 들어.

C는 학생이 수동적인 학습자에서 발음을 판단하고 가르쳐 주는 역할로 전환된다는 점에서, E는 발음의 차이가 실제 의미와 결과의 차이로 이어진다는 점에서 교육적 가치가 있어.

다만 E의 음성 인식 기능은 정밀한 발음 평가 도구가 아니므로 인식 실패를 학생의 발음 오류로 단정하지 않고, 재시도와 자기 점검이 가능하도록 설계해야 해.

그리고 모든 활동을 하나의 스토리로 연결하면 개별 미니게임을 모아 놓은 앱보다 몰입감이 더 높아질 것 같아. 예를 들어 영어 발음을 혼동하는 로봇을 도와 소리의 비밀을 하나씩 해결하는 모험으로 전체 앱을 구성하는 거지.

function escapeHTML(text) {
  return String(text)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}

function shuffle(items) {
  const result = [...items];

  for (let i = result.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [result[i], result[j]] = [result[j], result[i]];
  }

  return result;
}

function getEnglishVoice() {
  return (
    voiceCache.find(v => v.lang === "en-US" && v.localService) ||
    voiceCache.find(v => v.lang === "en-US") ||
    voiceCache.find(v => v.lang.startsWith("en")) ||
    null
  );
}

function speak(text, rate = 0.85) {
  if (!("speechSynthesis" in window)) {
    alert("이 브라우저에서는 모델 음성 재생을 지원하지 않아요.");
    return;
  }

  if (activeRecorder || recognizing) {
    alert("녹음 또는 음성 인식을 먼저 종료해 주세요.");
    return;
  }

  speechVersion++;
  speechSynthesis.cancel();

  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang = "en-US";
  utterance.rate = rate;
  utterance.pitch = 1;

  const voice = getEnglishVoice();

  if (voice) {
    utterance.voice = voice;
  }

  utterance.onerror = function(event) {
    if (
      event.error !== "canceled" &&
      event.error !== "interrupted"
    ) {
      alert("음성을 재생할 수 없어요. 기기의 영어 음성 설정을 확인해 주세요.");
    }
  };

  speechSynthesis.speak(utterance);
}

function speakSequence(words, rate = 0.8) {
  if (!("speechSynthesis" in window)) {
    alert("모델 음성을 사용할 수 없는 브라우저예요.");
    return;
  }

  if (activeRecorder || recognizing) {
    alert("녹음 또는 음성 인식을 먼저 종료해 주세요.");
    return;
  }

  speechSynthesis.cancel();

  const version = ++speechVersion;
  let index = 0;

  function playNext() {
    if (
      version !== speechVersion ||
      index >= words.length
    ) return;

    const utterance = new SpeechSynthesisUtterance(words[index++]);
    utterance.lang = "en-US";
    utterance.rate = rate;

    const voice = getEnglishVoice();

    if (voice) utterance.voice = voice;

    utterance.onend = function() {
      setTimeout(playNext, 550);
    };

    utterance.onerror = function() {
      if (version === speechVersion) {
        alert("음성을 재생할 수 없어요.");
      }
    };

    speechSynthesis.speak(utterance);
  }

  playNext();
}

async function startRecording(key) {
  const status = $("status-" + key);
  const startButton = $("start-" + key);

  if (activeRecorder || recognizing) {
    alert("진행 중인 마이크 활동을 먼저 종료해 주세요.");
    return;
  }

  if (
    !navigator.mediaDevices ||
    !navigator.mediaDevices.getUserMedia ||
    !window.MediaRecorder
  ) {
    if (status) {
      status.textContent =
        "이 환경에서는 녹음을 지원하지 않아요. HTTPS와 마이크 권한을 확인해 주세요.";
    }

    return;
  }

  const originSession = sessionId;
  const originView = currentView;
  let stream = null;

  if ("speechSynthesis" in window) {
    speechVersion++;
    speechSynthesis.cancel();
  }

  try {
    if (startButton) startButton.disabled = true;

    if (status) {
      status.textContent = "마이크 연결 중...";
    }

    stream = await navigator.mediaDevices.getUserMedia({
      audio: true
    });

    if (
      originSession !== sessionId ||
      originView !== currentView ||
      !$("status-" + key)
    ) {
      stream.getTracks().forEach(track => track.stop());
      return;
    }

    if (activeRecorder || recognizing) {
      stream.getTracks().forEach(track => track.stop());
      if (startButton) startButton.disabled = false;
      return;
    }

    const recorder = new MediaRecorder(stream);
    const chunks = [];

    const recordState = {
      recorder,
      stream,
      key,
      session: sessionId,
      timer: null
    };

    activeRecorder = recordState;

    recorder.ondataavailable = event => {
      if (event.data.size > 0) {
        chunks.push(event.data);
      }
    };

    recorder.onstop = () => {
      clearTimeout(recordState.timer);
      stream.getTracks().forEach(track => track.stop());

      if (activeRecorder === recordState) {
        activeRecorder = null;
      }

      if (recordState.session !== sessionId) return;

      const blob = new Blob(chunks, {
        type: recorder.mimeType || "audio/webm"
      });

      if (blob.size === 0) {
        if ($("status-" + key)) {
          $("status-" + key).textContent =
            "녹음된 소리가 없어요. 다시 시도해 주세요.";
        }
        return;
      }

      if (recordings[key]) {
        URL.revokeObjectURL(recordings[key].url);
      }

      recordings[key] = {
        blob,
        url: URL.createObjectURL(blob)
      };

      const audioContainer = $("audio-" + key);

      if (audioContainer) {
        audioContainer.innerHTML =
          `<audio controls src="${recordings[key].url}"></audio>`;
      }

      const statusElement = $("status-" + key);

      if (statusElement) {
        statusElement.textContent =
          "✓ 녹음 완료! 모델 음성과 비교해 보세요.";
        statusElement.classList.remove("recording-indicator");
      }

      const start = $("start-" + key);
      const stop = $("stop-" + key);

      if (start) {
        start.disabled = false;
        start.textContent = "🎙️ 다시 녹음";
      }

      if (stop) stop.disabled = true;
    };

    recorder.onerror = () => {
      stopRecording();

      if ($("status-" + key)) {
        $("status-" + key).textContent =
          "녹음 중 문제가 발생했어요. 다시 시도해 주세요.";
      }
    };

    recorder.start();

    if (status) {
      status.textContent =
        "🔴 녹음 중... 또박또박 말해 주세요!";
      status.classList.add("recording-indicator");
    }

    $("stop-" + key).disabled = false;

    recordState.timer = setTimeout(() => {
      if (activeRecorder === recordState) {
        stopRecording();
      }
    }, 15000);

  } catch (error) {
    if (stream) {
      stream.getTracks().forEach(track => track.stop());
    }

    if (startButton) startButton.disabled = false;

    if (status) {
      status.textContent =
        "마이크를 사용할 수 없어요. 권한을 확인하거나 녹음 없이 연습해 주세요.";
    }
  }
}

function stopRecording() {
  if (!activeRecorder) return;

  const item = activeRecorder;

  clearTimeout(item.timer);

  if (item.recorder.state !== "inactive") {
    item.recorder.stop();
  }

  item.stream.getTracks().forEach(track => track.stop());
  activeRecorder = null;
}

function stopMedia() {
  speechVersion++;

  if ("speechSynthesis" in window) {
    speechSynthesis.cancel();
  }

  if (activeRecorder) {
    stopRecording();
  }

  if (recognition) {
    const oldRecognition = recognition;
    recognition = null;
    recognizing = false;

    oldRecognition.onresult = null;
    oldRecognition.onerror = null;
    oldRecognition.onend = null;

    try {
      oldRecognition.abort();
    } catch {}
  }

  document.querySelectorAll("audio").forEach(audio => {
    audio.pause();
  });
}

function clearRecordings() {
  Object.values(recordings).forEach(item => {
    URL.revokeObjectURL(item.url);
  });

  recordings = {};
}

    alert("마이크 활동을 먼저 종료해 주세요.");

  }

  stage++;
  questionIndex = 0;

  switch (stage) {
    case 1:
      initDelivery();
      break;

    case 2:
      initDetective();
      break;

    case 3:
      renderStudio();
      break;

    case 4:
      renderAdventure();
      break;

    case 5:
      renderBattle();
      break;

function initDelivery() {
  questions = shuffle(
    lesson.pairs.flatMap((pair, index) => [
      { pair, target: 0, pairIndex: index },
      { pair, target: 1, pairIndex: index }
    ])
  ).slice(0, 6);

  questions.forEach(q => {
    q.options = shuffle(
      q.pair.map((item, i) => ({

        originalIndex: i

  });

  questionIndex = 0;

  renderDelivery();
}

            : "다음 배달 →"}

        </button>
      </div>

    "delivery"
  );

function playDelivery(slow) {
  const q = questions[questionIndex];
  const target = q.pair[q.target].word;

  if (!("speechSynthesis" in window)) {
    feedback(
      "deliveryFeedback",
      "음성 재생이 지원되는 기기에서 이 듣기 활동을 진행해 주세요.",
      "neutral"
    );
    return;
  }

  speak(target, slow ? 0.65 : 0.9);

  listened = true;

  if (!answered) {
    q.options.forEach((item, i) => {
      const button = $("delivery-" + i);

      if (button && !button.classList.contains("incorrect")) {
        button.disabled = false;
      }
    });
  }
}

function checkDelivery(index) {
  if (!listened || answered) return;

  const q = questions[questionIndex];
  const selected = q.options[index];

  if (selected.originalIndex === q.target) {

    answered = true;
    stats.listeningTotal++;

    if (!hadWrongAttempt) {
      stats.listeningCorrect++;
    }

    $("delivery-" + index).classList.add("correct");

    $("deliveryTruck").classList.add("delivered");

    q.options.forEach((item, i) => {
      $("delivery-" + i).disabled = true;
      $("deliveryWord-" + i).classList.remove("hidden");
    });

    const target = q.pair[q.target];

    feedback(
      "deliveryFeedback",
      `📦 배달 성공! 고객이 주문한 것은 ${target.word} (${target.meaning})였어요.`,
      "good"
    );

    speak(target.word, 0.8);
    addStar();

    $("deliveryNext").classList.remove("hidden");

  } else {

    hadWrongAttempt = true;

    $("delivery-" + index).classList.add("incorrect");
    $("delivery-" + index).disabled = true;

    feedback(
      "deliveryFeedback",
      "앗, 손님이 주문한 물건이 아니에요. 다시 듣고 두 소리의 차이를 찾아보세요!",
      "retry"
    );
  }
}

function nextDelivery() {
  if (!answered) return;

  questionIndex++;

  if (questionIndex >= questions.length) {
    nextStage();
  } else {
    renderDelivery();
  }
}

  const reallyCorrect = !q.isMistake;

  if (userSaysCorrect === reallyCorrect) {

    answered = true;
    stats.detectiveTotal++;

    if (!hadWrongAttempt) {
      stats.detectiveCorrect++;
    }

    $("detectiveYes").disabled = true;
    $("detectiveNo").disabled = true;

    feedback(
      "detectiveFeedback",

      q.isMistake
        ? `🔍 사건 해결! ROBO는 ${q.pair[q.spoken].word}라고 말했어요. 의도한 단어는 ${q.pair[q.target].word}였어요.`
        : `🎉 정확히 알아들었어요! ROBO가 의도한 단어를 말했어요.`,

      "good"
    );

    $("correctionPanel").classList.remove("hidden");
    addStar();

  } else {

    hadWrongAttempt = true;

    feedback(
      "detectiveFeedback",
      "아직 사건이 해결되지 않았어요. 그림 속 단어와 ROBO가 말한 소리를 다시 비교해 보세요.",
      "retry"
    );
  }
}

function startRecognition() {

  if (answered || recognizing) return;

  if (activeRecorder) {
    alert("녹음을 먼저 종료해 주세요.");
    return;
  }

  if (!$("recognitionConsent").checked) {
    feedback(
      "adventureFeedback",
      "음성 인식 사용 동의를 확인하거나 직접 확인 모드를 이용해 주세요.",
      "neutral"
    );
    return;
  }

  const SpeechRecognition =
    window.SpeechRecognition ||
    window.webkitSpeechRecognition;

  if (!SpeechRecognition) {

    feedback(
      "adventureFeedback",
      "이 브라우저는 음성 인식을 지원하지 않아요. 직접 확인 모드로 미션을 진행할 수 있어요.",
      "neutral"
    );

    showManualMission();
    return;
  }

  if ("speechSynthesis" in window) {
    speechVersion++;
    speechSynthesis.cancel();
  }

  const instance = new SpeechRecognition();

  instance.lang = "en-US";
  instance.interimResults = false;
  instance.continuous = false;
  instance.maxAlternatives = 3;

  recognition = instance;
  recognizing = true;

  $("recognizeButton").disabled = true;
  $("stopRecognizeButton").disabled = false;

  $("recognitionStatus").textContent =
    "마이크 연결 중...";

  $("transcript").textContent =
    "🎙️ Listening...";

  const originSession = sessionId;
  const originView = currentView;

  instance.onstart = () => {
    if (recognition !== instance) return;

    $("recognitionStatus").textContent =
      "지금 영어로 말해 주세요!";
  };

  instance.onresult = event => {

    if (
      recognition !== instance ||
      originSession !== sessionId ||
      originView !== currentView
    ) return;

    const transcript =
      event.results[0][0].transcript;

    $("transcript").textContent =
      'ROBO heard: "' + transcript + '"';

    evaluateMissionSpeech(transcript);
  };

  instance.onerror = event => {

    if (recognition !== instance) return;

    const messages = {
      "no-speech": "음성을 듣지 못했어요. 다시 말해 주세요.",
      "not-allowed": "마이크 권한이 허용되지 않았어요.",
      "audio-capture": "마이크를 찾을 수 없어요.",
      "network": "음성 인식 연결에 문제가 발생했어요."
    };

    if (event.error !== "aborted") {
      feedback(
        "adventureFeedback",
        messages[event.error] ||
        "음성 인식에 문제가 발생했어요. 다시 시도해 주세요.",
        "neutral"
      );
    }

    if (
      event.error === "not-allowed" ||
      event.error === "network" ||
      event.error === "audio-capture"
    ) {
      $("manualMission").classList.remove("hidden");
    }
  };

  instance.onend = () => {

    if (recognition !== instance) return;

    recognition = null;
    recognizing = false;

    if ($("recognizeButton")) {
      $("recognizeButton").disabled = answered;
      $("stopRecognizeButton").disabled = true;

      $("recognitionStatus").textContent =
        answered
          ? "미션 전달 확인 완료!"
          : "다시 말하거나 직접 확인할 수 있어요.";
    }
  };

  try {
    instance.start();
  } catch {

    recognition = null;
    recognizing = false;

    $("recognizeButton").disabled = false;
    $("stopRecognizeButton").disabled = true;

    feedback(
      "adventureFeedback",
      "음성 인식을 시작할 수 없어요. 직접 확인 모드를 이용해 주세요.",
      "neutral"
    );

    $("manualMission").classList.remove("hidden");
  }
}

function stopRecognition() {
  if (!recognition) return;

  try {
    recognition.stop();
  } catch {}
}

function evaluateMissionSpeech(transcript) {

  if (answered) return;

  const mission = lesson.missions[questionIndex];

  const words = transcript
    .toLowerCase()
    .replace(/[^a-z\s]/g, " ")
    .split(/\s+/)
    .filter(Boolean);

  const matches = mission.pair
    .map((item, index) => ({
      index,
      found: words.includes(item.word.toLowerCase())
    }))
    .filter(item => item.found);

  document.querySelectorAll(".adventure-item").forEach(item => {
    item.classList.remove("selected");
  });

  if (matches.length !== 1) {

    feedback(
      "adventureFeedback",
      "ROBO가 어떤 단어를 선택해야 할지 모르겠대요. 필요한 단어 하나를 천천히 다시 말해 보세요.",
      "neutral"
    );

    return;
  }

  const recognizedIndex = matches[0].index;

  $("adventureItem-" + recognizedIndex)
    .classList.add("selected");

  if (recognizedIndex === mission.target) {

    missionRecognized = true;

    finishMission(true);

  } else {

    const actual = mission.pair[recognizedIndex];
    const desired = mission.pair[mission.target];

    feedback(
      "adventureFeedback",
      `ROBO는 ${actual.word} (${actual.meaning})로 알아들었어요. 원하는 것은 ${desired.meaning}이었죠? 두 소리를 비교하고 다시 도전해 보세요.`,
      "retry"
    );

    $("missionHint").classList.remove("hidden");
  }
}

function manualMissionAnswer(index) {

  if (answered) return;

  if (activeRecorder || recognizing) {
    alert("마이크 활동을 먼저 종료해 주세요.");
    return;
  }

  const mission = lesson.missions[questionIndex];

  document.querySelectorAll(".adventure-item").forEach(item => {
    item.classList.remove("selected");
  });

  $("adventureItem-" + index).classList.add("selected");

  if (index === mission.target) {

    $("transcript").textContent =
      "직접 선택: " + mission.pair[index].word;

    finishMission(false);

  } else {

    feedback(
      "adventureFeedback",
      "지금 필요한 물건이나 그림과 선택한 단어의 의미를 다시 비교해 보세요.",
      "retry"
    );
  }
}

function finishMission(usedRecognition) {

  if (answered) return;

  answered = true;

  stats.missionsCompleted++;

  if (usedRecognition) {
    stats.recognizedMissions++;
  }

  addStar();

  const mission = lesson.missions[questionIndex];

  const target = mission.pair[mission.target];

  feedback(
    "adventureFeedback",

    usedRecognition
      ? `🚀 미션 성공! ROBO가 ${target.word}를 인식했어요. 원하는 의미가 전달되었네요!`
      : `🚀 미션 완료! ${target.word}를 올바르게 선택했어요. 직접 확인 모드로 모험을 마쳤어요.`,

    "good"
  );

  $("recognizeButton").disabled = true;

  $("adventureNext").classList.remove("hidden");

  $("adventureRobot").style.transform =
    mission.target === 0
      ? "translateX(-25px)"
      : "translateX(25px)";
}

  } else {

function finishQuest() {

  if (activeRecorder) {
    alert("마지막 녹음을 먼저 종료해 주세요.");
    return;
  }

  const choice = document.querySelector(
    'input[name="confidence"]:checked'
  );

  if (!choice) {
    alert("오늘의 자신감 상태를 하나 선택해 주세요.");
    return;
  }

  stats.confidence = Number(choice.value);

  stats.reflection = [
    $("reflect1").checked,
    $("reflect2").checked,
    $("reflect3").checked,
    $("reflect4").checked
  ];

  addStar();

  completed[currentKey] = true;
  writeStorage("soundQuestCompleted", completed);

  stage = 6;

  renderComplete();
}
